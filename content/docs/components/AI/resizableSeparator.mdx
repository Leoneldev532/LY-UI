## Example

```tsx
import { ResizableSeparator } from '@/components/LY UI/AI/resizableSeparator'
import { useRef, useState } from 'react'

export default function SeparatorExample() {
  const [leftSize, setLeftSize] = useState(200)
  const parentRef = useRef<HTMLDivElement>(null)

  const handleResize = (newSize: number) => {
    setLeftSize(newSize)
  }

  return (
    <div ref={parentRef} className="flex h-64 border rounded-lg">
      <div 
        className="bg-blue-100 dark:bg-blue-900/20 p-4"
        style={{ width: leftSize }}
      >
        <p>Left zone</p>
      </div>
      
      <ResizableSeparator
        direction="horizontal"
        variant="outline"
        size="medium"
        onResize={handleResize}
        parentRef={parentRef}
        showIndicator={true}
        minSize={100}
        maxSize={400}
        animated={true}
        zIndex={20}
      />
      
      <div className="flex-1 bg-green-100 dark:bg-green-900/20 p-4">
        <p>Right zone</p>
      </div>
    </div>
  )
}
```

## Documentation

| Prop | Type | Default | Description |
|------|------|---------|-------------|
| `direction` | `"vertical" \| "horizontal"` | `"vertical"` | Separator direction |
| `variant` | `"ghost" \| "outline"` | `"ghost"` | Visual style of the separator |
| `size` | `"small" \| "medium" \| "large"` | `"medium"` | Size of the separator |
| `onResize` | `(size: number) => void` | - | Callback called during resizing (required) |
| `minSize` | `number` | `100` | Minimum size in pixels |
| `maxSize` | `number` | `600` | Maximum size in pixels |
| `parentRef` | `RefObject<HTMLElement \| null>` | - | Reference to parent element (required) |
| `showIndicator` | `boolean` | `false` | Shows visual indicator |
| `animated` | `boolean` | `true` | Enables animations |
| `zIndex` | `number` | `20` | Z-index of the separator |
| `className` | `string` | - | Additional CSS classes |

## Features

- **Drag and drop** : Intuitive resizing by drag and drop
- **Multiple directions** : Vertical and horizontal support
- **Configurable limits** : Customizable minimum and maximum sizes
- **Visual indicators** : Optional indicator bar
- **Smooth animations** : Smooth transitions during resizing
- **Responsive** : Adapts to all screen sizes
- **Accessibility** : Screen reader support
- **Touch support** : Compatible with touch devices
