## Example

```tsx
import { TagButton } from '@/components/LY UI/AI/tagButton'
import { SearchSlashIcon, Code, Image, Bot } from 'lucide-react'

export default function TagExample() {
  const tools = [
    "Web Search",
    "Image Generation", 
    "Code Analysis",
    "AI Assistant"
  ]

  const handleItemClick = (item: string) => {
    console.log('Selected tool:', item)
  }

  const handleClick = () => {
    console.log('Button clicked')
  }

  return (
    <div className="flex gap-4">
      <TagButton
        title="AI Tools"
        variant="default"
        icon={<SearchSlashIcon className="w-4 h-4" />}
        items={tools}
        onItemClick={handleItemClick}
        onClick={handleClick}
        className="px-4 py-2"
      />

      <TagButton
        title="Models"
        variant="outline"
        icon={<Bot className="w-4 h-4" />}
        items={["GPT-4", "Claude-3", "Llama-2"]}
        onItemClick={(model) => console.log('Model:', model)}
      />
    </div>
  )
}
```

## Documentation

| Prop | Type | Default | Description |
|------|------|---------|-------------|
| `title` | `string` | - | Text displayed on the button (required) |
| `variant` | `"default" \| "outline" \| "ghost"` | - | Visual style of the button (required) |
| `icon` | `React.ReactNode` | - | Icon to display next to the text |
| `onClick` | `() => void` | - | Callback called when button is clicked |
| `items` | `string[]` | `[]` | List of items to display in dropdown |
| `onItemClick` | `(item: string) => void` | - | Callback called when an item is selected |
| `className` | `string` | - | Additional CSS classes |

## Features

- **Smooth animations** : Smooth transitions for dropdown open/close
- **Icon support** : Ability to add an icon to the button
- **Animated dropdown** : List of items with appearance animations
- **State management** : Only one dropdown open at a time
- **Visual variants** : Three different styles (default, outline, ghost)
- **Responsive** : Adapts to all screen sizes
- **Accessibility** : Screen reader support
