## Example

```tsx
import { StepProgressTimeline } from '@/components/LY UI/AI/stepProgressTimeline'

export default function TimelineExample() {
  const steps = [
    { step: 1, stepName: "Initialization", duration: 2000, status: "waiting..." },
    { step: 2, stepName: "Analysis", duration: 3000, status: "waiting..." },
    { step: 3, stepName: "Processing", duration: 2500, status: "waiting..." },
    { step: 4, stepName: "Finalization", duration: 1500, status: "waiting..." }
  ]

  const handleGenerationStart = () => {
    console.log('Generation started')
  }

  const handleGenerationComplete = () => {
    console.log('Generation completed')
  }

  const handleGenerationStop = () => {
    console.log('Generation stopped')
  }

  return (
    <StepProgressTimeline
      steps={steps}
      onGenerationStart={handleGenerationStart}
      onGenerationComplete={handleGenerationComplete}
      onGenerationStop={handleGenerationStop}
      externalStart={true}
      externalStop={false}
      scrollSpeed="normal"
      className="w-full max-w-md"
    />
  )
}
```

## Documentation

| Prop | Type | Default | Description |
|------|------|---------|-------------|
| `steps` | `StepProps[]` | - | Array of steps to display (required) |
| `onGenerationStart` | `() => void` | - | Callback called when generation starts |
| `onGenerationComplete` | `() => void` | - | Callback called when generation completes |
| `onGenerationStop` | `() => void` | - | Callback called when generation stops |
| `externalStop` | `boolean` | `false` | External control to stop generation |
| `externalStart` | `boolean` | `false` | External control to start generation |
| `className` | `string` | - | Additional CSS classes |
| `scrollSpeed` | `"fast" \| "normal" \| "slow"` | `"normal"` | Timeline scroll speed |

## Types

```tsx
type StepProps = {
  step?: number;
  stepName?: string;
  duration?: number;
  status: "waiting..." | "generating..." | "completed";
};
```

## Features

- **Sequential animations** : Automatic progression from one step to another
- **Status indicators** : Visual display of each step's state
- **External controls** : Ability to start/stop from outside
- **Complete callbacks** : Notifications for all generation events
- **Configurable timing** : Customizable duration for each step
- **Responsive** : Adapts to all screen sizes
- **Accessibility** : Screen reader support
