## Example

```tsx
export default function AnimatedProcessListExample() {
  const [isAnimating, setIsAnimating] = useState(false)
  const [animationSpeed, setAnimationSpeed] = useState<'slow' | 'normal' | 'fast'>('normal')
  const [scrollDirection, setScrollDirection] = useState<'up' | 'down'>('up')

  const handleAnimationEnd = () => {
    console.log('Process list animation completed')
    setIsAnimating(false)
  }

  const handleStartAnimation = () => {
    setIsAnimating(true)
  }

  const processSteps = [
    "Initializing development environment and dependencies",
    "Setting up project structure and configuration files",
    "Creating reusable components and utility functions",
    "Implementing responsive design and mobile optimization",
    "Adding interactive features and state management",
    "Testing functionality across different browsers",
    "Optimizing bundle size and loading performance",
    "Deploying to production environment",
  ]

  return (
    <div className="space-y-6">
      <div className="flex gap-4 items-center">
        <button
          onClick={handleStartAnimation}
          className="px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700"
        >
          Start Process
        </button>
        <select
          value={animationSpeed}
          onChange={(e) => setAnimationSpeed(e.target.value as 'slow' | 'normal' | 'fast')}
          className="px-3 py-2 border rounded-lg"
        >
          <option value="slow">Slow</option>
          <option value="normal">Normal</option>
          <option value="fast">Fast</option>
        </select>
        <select
          value={scrollDirection}
          onChange={(e) => setScrollDirection(e.target.value as 'up' | 'down')}
          className="px-3 py-2 border rounded-lg"
        >
          <option value="up">Scroll Up</option>
          <option value="down">Scroll Down</option>
        </select>
      </div>

      <AnimatedProcessList
        title="Building your application..."
        processSteps={processSteps}
        animationSpeed={animationSpeed}
        scrollDirection={scrollDirection}
        onAnimationEnd={handleAnimationEnd}
        skeletonDuration={1500}
        lineRevealDelay={200}
        className="w-full max-w-lg"
      />
    </div>
  )
}
```

## Documentation

| Prop | Type | Default | Description |
|------|------|---------|-------------|
| `title` | `string` | `"Generating elements for your web page"` | Title displayed at the top of the component |
| `processSteps` | `string[]` | `[]` | Array of process steps to display in the animated list |
| `className` | `string` | `undefined` | Additional CSS classes for styling |
| `scrollDirection` | `"up" \| "down"` | `"up"` | Direction of the scrolling animation |
| `animationSpeed` | `"slow" \| "normal" \| "fast"` | `"normal"` | Speed of the animation |
| `onAnimationEnd` | `() => void` | `undefined` | Callback when the reveal animation ends |
| `skeletonDuration` | `number` | `2000` | Duration in milliseconds of the skeleton phase |
| `lineRevealDelay` | `number` | `150` | Delay in milliseconds between each line reveal |

### Types

```tsx
interface AnimatedProcessListProps {
  title?: string
  processSteps?: string[]
  className?: string
  scrollDirection?: "up" | "down"
  animationSpeed?: "slow" | "normal" | "fast"
  onAnimationEnd?: () => void
  skeletonDuration?: number
  lineRevealDelay?: number
}
```

## Features

- **Skeleton Loading**: Initial loading state with animated skeleton placeholders
- **Progressive Reveal**: Lines appear one by one with smooth fade-in animations
- **Continuous Scrolling**: Infinite scroll animation with customizable direction and speed
- **Line Numbering**: Automatic line numbering for process steps
- **Responsive Design**: Adapts to different screen sizes with proper scaling
- **Dark Mode Support**: Full dark mode compatibility with appropriate color schemes
- **Customizable Timing**: Configurable skeleton duration and line reveal delays
- **Accessibility Focused**: Screen reader friendly with proper semantic structure
