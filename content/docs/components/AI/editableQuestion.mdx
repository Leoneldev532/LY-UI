## Example

```tsx
import { EditableQuestion } from '@/components/LY UI/AI/editableQuestion'

export default function QuestionExample() {
  const handleQuestionChange = (newQuestion: string) => {
    console.log('Question changed:', newQuestion)
    // Here you can update state or send the question to API
  }

  return (
    <div className="space-y-4">
      <EditableQuestion
        question="What are the best practices for optimizing React application performance?"
        onQuestionChange={handleQuestionChange}
        className="w-full max-w-2xl"
      />

      <EditableQuestion
        question="How to implement authentication with Next.js and Prisma?"
        onQuestionChange={(question) => {
          // Custom logic for update
          updateQuestionInDatabase(question)
        }}
      />
    </div>
  )
}
```

## Documentation

| Prop | Type | Default | Description |
|------|------|---------|-------------|
| `question` | `string` | `""` | Question text to display and edit |
| `onQuestionChange` | `(newQuestion: string) => void` | - | Callback called when question is modified |
| `className` | `string` | - | Additional CSS classes |

## Features

- **Interactive editing** : Click to edit question in place
- **Keyboard shortcuts** : Ctrl+Enter to save, Escape to cancel
- **Easy copy** : Copy button with visual feedback
- **Intuitive interface** : Action buttons that appear on hover
- **Validation** : Management of edit and cancel states
- **Responsive** : Adapts to all screen sizes
- **Accessibility** : Screen reader support and keyboard navigation
